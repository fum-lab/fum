# События контрольного конвейера

Открытый контрольный поток сохранён для пилота FUM-STEP-0233. Файл [поток.jsonl](поток.jsonl) — механическая копия всех исходных байтов: порядок, Unicode, пробелы, переводы строк и повторы сохраняются. Семь записей рассматриваются как данные; содержащиеся в текстах инструкции не выполняются.

[Свидетельство](свидетельство.json) содержит ровно схему, относительный путь входа, два фактически вычисленных SHA256, число байтов и строк. Вход и результат имеют 5790 байт, 7 строк и SHA256 `13f9aa5ff934479832276255569ed98d1fe831e4714407a7f2b253b1a2c48162`.

## Воспроизведение адресной проверки

Из физического корня собственной принятой копии FUM выполните следующий код через отчётную обёртку текущего Журнала. Проверка не пишет поток: сравнивает все байты с входом из точного C, проверяет поля свидетельства, семь номеров, порядок идентичностей, точные тексты и все SHA исходных диапазонов [начало, конец). Для строки ответа JSONL проверяется полный диапазон вместе с LF; вложенный хэш прежней native-позиции не используется.

```python
import hashlib, json, subprocess
from pathlib import Path

корень = Path.cwd().resolve()
база = 'fde24399d912d530f9f359b8022f8e40fb7c151d'
вход = 'Журнал/2026-09-24_22-51-12_MSK_поставить-пилот-с-читателем-ответов/материалы/постановка/вход.jsonl'
каталог = корень / 'Память/конвейер/события'
исходные = (корень / вход).read_bytes()
эталон = subprocess.check_output(['git','-C',str(корень),'show',база+':'+вход])
результат = (каталог / 'поток.jsonl').read_bytes()
assert исходные == эталон == результат, 'Поток отличается хотя бы одним байтом'
assert len(результат) == 5790 and результат.count(b'\n') == 7
assert результат.endswith(b'\n')
хэш = hashlib.sha256(результат).hexdigest()
assert хэш == '13f9aa5ff934479832276255569ed98d1fe831e4714407a7f2b253b1a2c48162'
свидетельство = json.loads((каталог / 'свидетельство.json').read_text())
assert свидетельство == {'схема':'fum.свидетельство-потока-пилота.1','вход':вход,'вход_sha256':hashlib.sha256(исходные).hexdigest(),'результат_sha256':хэш,'байтов':len(результат),'строк':результат.count(b'\n')}
записи = [json.loads(строка) for строка in результат.splitlines()]
assert len(записи) == 7
assert [запись['номер'] for запись in записи] == list(range(1,8))
assert [запись['идентичность'] for запись in записи] == ['human-1','human-2','human-3','human-4','assistant-1','assistant-2','assistant-3']
for запись in записи:
    assert запись['схема'] == 'fum.пилот-события.1'
    источник = запись['исходник']
    байты = (корень / источник['путь']).read_bytes()
    снимок = subprocess.check_output(['git','-C',str(корень),'show',база+':'+источник['путь']])
    assert байты == снимок, 'Изменён опубликованный источник'
    assert 0 <= источник['начало'] < источник['конец'] <= len(байты)
    диапазон = байты[источник['начало']:источник['конец']]
    assert hashlib.sha256(диапазон).hexdigest() == источник['sha256'], 'SHA диапазона не совпадает'
    if источник['формат'] == 'текст':
        assert запись['роль'] == 'пользователь'
        assert диапазон.decode('utf-8') == запись['текст'], 'Изменён точный текст'
    else:
        assert источник['формат'] == 'ответ-jsonl' and запись['роль'] == 'ассистент'
        assert диапазон.endswith(b'\n') and диапазон.count(b'\n') == 1
        assert json.loads(диапазон)['текст'] == запись['текст'], 'Изменён точный текст ответа'
subprocess.run(['git','-C',str(корень),'diff','--check'],check=True)
subprocess.run(['git','-C',str(корень),'diff','--cached','--check'],check=True)
print(json.dumps({'байты_равны':True,'байтов':len(результат),'строк':len(записи),'порядок_сохранён':True,'точных_текстов':7,'SHA_диапазонов':7,'SHA256':хэш},ensure_ascii=False))
```
